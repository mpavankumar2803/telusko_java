package com.telusko.jdbclearning;
import java.sql.*;
public class LaunchApp1 {

	public static void main(String[] args) throws ClassNotFoundException, SQLException {
		// load and register the driver
		Class.forName("com.mysql.cj.jdbc.Driver");
		
		// Establish connection 
		String url = "jdbc:mysql://localhost:3306/jdbclearning";
		String user="root";
		String password="pavan@2803";
		Connection con=DriverManager.getConnection(url,user,password);
		
		//creating statement
		Statement st = con.createStatement();
		
		//execute query
		String sql="SELECT * FROM studentinfo";
		
		ResultSet rs=st.executeQuery(sql);
		while(rs.next()){
		System.out.println(rs.getInt("id")+" "+rs.getString("sname")+" "+rs.getInt("sage")+" "+rs.getString("scity"));
		}
		//close connection
		rs.close();
		st.close();
		con.close();
	}

}

//////////////  below program changes the age according to the id   ////////////////////
package com.telusko.jdbclearning;
import java.util.*;
import java.sql.*;
public class LaunchApp1 {

	public static void main(String[] args){
		
		try(Connection con=DriverManager.getConnection("jdbc:mysql://localhost:3306/jdbclearning","root","pavan@2803");
				PreparedStatement ps=con.prepareStatement("UPDATE studentinfo SET sage=? WHERE id=?");
						){
			Class.forName("com.mysql.cj.jdbc.Driver");
			ps.setInt(1, 21);
			ps.setInt(2, 1);
			ps.addBatch();
			ps.setInt(1, 22);
			ps.setInt(2, 3);
			ps.addBatch();
			ps.setInt(1, 24);
			ps.setInt(2, 4);
			ps.addBatch();
			
			ps.executeBatch();
			System.out.println("check the db table to see the result");
			
		}catch(Exception e) {
			e.printStackTrace();
		}
	}
}
